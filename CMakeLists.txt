cmake_minimum_required(VERSION 3.16)
project(ChoirOfRuin VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CHOIR_SANITISE "Enable address and undefined behaviour sanitisers" OFF)
option(CHOIR_BUILD_APP "Build the SDL2 graphical game" ON)
add_library(choir_engine src/game.cpp)
target_include_directories(choir_engine PUBLIC src)
if(MSVC)
  target_compile_options(choir_engine PUBLIC /W4)
else()
  target_compile_options(choir_engine PUBLIC -Wall -Wextra -Wpedantic)
  if(CHOIR_SANITISE)
    target_compile_options(choir_engine PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(choir_engine PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(CHOIR_BUILD_APP)
  find_package(SDL2 CONFIG REQUIRED)
  add_executable(choir src/main.cpp src/view.cpp)
  target_link_libraries(choir PRIVATE choir_engine SDL2::SDL2)
  if(TARGET SDL2::SDL2main)
    target_link_libraries(choir PRIVATE SDL2::SDL2main)
  endif()
  add_executable(choir_view_tests tests/view_tests.cpp src/view.cpp)
  target_link_libraries(choir_view_tests PRIVATE choir_engine SDL2::SDL2)
  target_compile_definitions(choir_view_tests PRIVATE SDL_MAIN_HANDLED)
endif()
enable_testing()
add_executable(choir_tests tests/game_tests.cpp)
target_link_libraries(choir_tests PRIVATE choir_engine)
add_test(NAME engine_contracts COMMAND choir_tests)
if(CHOIR_BUILD_APP)
  add_test(NAME pixel_renderer COMMAND choir_view_tests)
endif()
